# Mong Samian ERP Sync Schedule
# These entries are merged into the host crontab alongside LEBAZAI jobs.
# To re-install after edits, manually append to: crontab -e
#
# Times chosen to avoid collision with LEBAZAI's Samian jobs (02:30 full, 13:40 prices).

# Combined price + barcodes + inventory: every 15 min during business hours
*/15 9-22 * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_sync.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py sync_samian_erp >> /mnt/sda2/mong_samian/logs/sync.log 2>&1

# Full sweep (not incremental): daily at 2:45 AM
45 2 * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_sync_full.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py sync_samian_erp --full >> /mnt/sda2/mong_samian/logs/sync_full.log 2>&1

# Full catalog (products, brands, departments, colors): daily at 3:15 AM
15 3 * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_sync_catalog.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py sync_samian_catalog >> /mnt/sda2/mong_samian/logs/sync_catalog.log 2>&1

# Build compressed WebP images (Samian galleries + legacy site media) into media/img: daily at 3:45 AM,
# after the catalog/colors sync. Only new images are built; --prune removes images no longer used.
45 3 * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_build_images.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py build_images --prune >> /mnt/sda2/mong_samian/logs/build_images.log 2>&1

# Site customers from Samian (customer, addresses, invoices, invoice rows): daily at 4:30 AM
30 4 * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_sync_customers.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py sync_samian_customers >> /mnt/sda2/mong_samian/logs/sync_customers.log 2>&1

# Mong sales (all branches, Lebaza's included) for best sellers and purchase history:
# new rows every hour during business hours, full re-read nightly at 4:15 AM.
5 9-23 * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_sync_sales.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py sync_samian_sales >> /mnt/sda2/mong_samian/logs/sync_sales.log 2>&1
15 4 * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_sync_sales.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py sync_samian_sales --full >> /mnt/sda2/mong_samian/logs/sync_sales.log 2>&1

# Payment hub: asks SnappPay about unclear payments and refunds what was never confirmed
*/5 * * * * cd /mnt/sda2/mong_samian && flock -n /tmp/mong_payhub.lock /mnt/sda2/mong_samian/.venv/bin/python manage.py payhub_reconcile >> /mnt/sda2/mong_samian/logs/payhub.log 2>&1
